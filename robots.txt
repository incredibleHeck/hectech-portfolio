# The PDFs must stay crawlable. They are served with
# "X-Robots-Tag: noindex, noarchive", and a crawler has to be able to
# FETCH a file to see that header. Disallowing them here would block the
# fetch, the noindex would never be read, and a linked URL could still
# surface in results as a bare link. Allow the crawl; the header does the work.
User-agent: *
Allow: /
